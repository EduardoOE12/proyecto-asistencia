import React, { useState, useEffect, useRef } from 'react';
import { BrowserRouter, Routes, Route, Navigate, useNavigate } from 'react-router-dom';

// -------------------------------------------------------------
// COMPONENTE 1: Pantalla de Login
// -------------------------------------------------------------
const Login = () => {
  const [rol, setRol] = useState('Alumno');
  const [identificador, setIdentificador] = useState('');
  const [password, setPassword] = useState('');
  const [error, setError] = useState('');
  const navigate = useNavigate();

  const handleLogin = async (e) => {
    e.preventDefault();
    setError('');

    try {
      const response = await fetch('http://localhost:8000/api/login', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ rol, identificador, password })
      });

      const data = await response.json();

      if (!response.ok) {
        throw new Error(data.detail || 'Error al iniciar sesión');
      }

      // Guardar datos de sesión
      localStorage.setItem('usuario_identificador', identificador);
      localStorage.setItem('usuario_rol', data.rol);

      if (data.rol === 'Alumno') {
        navigate('/alumnos');
      } else {
        alert(`Sesión iniciada como ${data.rol}`);
      }
    } catch (err) {
      setError(err.message);
    }
  };

  return (
    <div style={{ maxWidth: '400px', margin: '50px auto', padding: '20px', border: '1px solid #ccc', borderRadius: '8px' }}>
      <h2>Iniciar Sesión</h2>
      {error && <p style={{ color: 'red' }}>{error}</p>}
      <form onSubmit={handleLogin}>
        <div style={{ marginBottom: '10px' }}>
          <label>Rol: </label>
          <select value={rol} onChange={(e) => setRol(e.target.value)} style={{ width: '100%', padding: '8px' }}>
            <option value="Alumno">Alumno</option>
            <option value="Docente">Docente</option>
            <option value="Director">Director</option>
          </select>
        </div>
        <div style={{ marginBottom: '10px' }}>
          <label>Matrícula / Correo: </label>
          <input
            type="text"
            value={identificador}
            onChange={(e) => setIdentificador(e.target.value)}
            required
            style={{ width: '100%', padding: '8px' }}
          />
        </div>
        <div style={{ marginBottom: '10px' }}>
          <label>Contraseña: </label>
          <input
            type="password"
            value={password}
            onChange={(e) => setPassword(e.target.value)}
            required
            style={{ width: '100%', padding: '8px' }}
          />
        </div>
        <button type="submit" style={{ width: '100%', padding: '10px', background: '#007bff', color: '#fff', border: 'none', borderRadius: '4px' }}>
          Ingresar
        </button>
      </form>
    </div>
  );
};

// -------------------------------------------------------------
// COMPONENTE 2: Pantalla de Alumnos (Consulta + Cámara)
// -------------------------------------------------------------
const Alumnos = () => {
  const [alumno, setAlumno] = useState(null);
  const [error, setError] = useState('');
  
  const [estadoCamara, setEstadoCamara] = useState('inicial'); // 'inicial', 'activa', 'bloqueada'
  const [opcionPermiso, setOpcionPermiso] = useState(null); // 'una_vez', 'en_uso', 'nunca'
  
  const videoRef = useRef(null);
  const streamRef = useRef(null);

  // 1. Consulta al backend de los datos del alumno
  useEffect(() => {
    const identificador = localStorage.getItem('usuario_identificador');

    if (!identificador) {
      setError('No se encontró una sesión activa.');
      return;
    }

    fetch(`http://localhost:8000/api/alumnos/consulta/${identificador}`)
      .then((res) => {
        if (!res.ok) throw new Error('Error al obtener datos del alumno');
        return res.json();
      })
      .then((data) => {
        if (data.ok) setAlumno(data.alumno);
      })
      .catch((err) => setError(err.message));
  }, []);

  // 2. Comprobación previa de permisos
  useEffect(() => {
    const permisoGuardado = localStorage.getItem('permiso_camara_modo');
    
    if (permisoGuardado === 'nunca') {
      setEstadoCamara('bloqueada');
      setOpcionPermiso('nunca');
    } else if (permisoGuardado === 'en_uso') {
      activarCamara('en_uso');
    }
  }, []);

  const activarCamara = async (modo) => {
    try {
      const stream = await navigator.mediaDevices.getUserMedia({ video: true, audio: false });
      streamRef.current = stream;

      if (videoRef.current) {
        videoRef.current.srcObject = stream;
      }

      setEstadoCamara('activa');
      setOpcionPermiso(modo);

      if (modo === 'en_uso') {
        localStorage.setItem('permiso_camara_modo', 'en_uso');
      }
    } catch (err) {
      console.error('Acceso denegado:', err);
      setEstadoCamara('bloqueada');
    }
  };

  const seleccionarPermiso = (opcion) => {
    if (opcion === 'nunca') {
      setEstadoCamara('bloqueada');
      setOpcionPermiso('nunca');
      localStorage.setItem('permiso_camara_modo', 'nunca');

      if (streamRef.current) {
        streamRef.current.getTracks().forEach((track) => track.stop());
      }
      return;
    }

    activarCamara(opcion);
  };

  useEffect(() => {
    return () => {
      const modo = localStorage.getItem('permiso_camara_modo');
      if (modo !== 'en_uso' && streamRef.current) {
        streamRef.current.getTracks().forEach((track) => track.stop());
      }
    };
  }, []);

  return (
    <div style={{ padding: '30px', maxWidth: '700px', margin: '0 auto', fontFamily: 'Arial, sans-serif' }}>
      <h2>Sección de Alumnos</h2>

      {error && <p style={{ color: 'red' }}>{error}</p>}
      
      {alumno ? (
        <div style={{ background: '#f8f9fa', border: '1px solid #ddd', padding: '15px', borderRadius: '8px', marginBottom: '20px' }}>
          <h3>Información del Alumno</h3>
          <p><strong>Nombre:</strong> {alumno.nombre}</p>
          <p><strong>Matrícula:</strong> {alumno.matricula}</p>
          <p><strong>Teléfono:</strong> {alumno.telefono}</p>
        </div>
      ) : (
        !error && <p>Cargando información del alumno...</p>
      )}

      <div style={{ border: '1px solid #ccc', padding: '20px', borderRadius: '8px', textAlign: 'center' }}>
        <h3>Acceso a la Cámara</h3>

        {estadoCamara === 'inicial' && (
          <div>
            <p>Selecciona una opción de permiso para la cámara:</p>
            <div style={{ display: 'flex', gap: '10px', justifyContent: 'center', marginTop: '15px' }}>
              <button onClick={() => seleccionarPermiso('una_vez')} style={{ padding: '10px 15px', cursor: 'pointer' }}>
                Solo esta vez
              </button>
              <button onClick={() => seleccionarPermiso('en_uso')} style={{ padding: '10px 15px', cursor: 'pointer' }}>
                Mientras la App esté en uso
              </button>
              <button 
                onClick={() => seleccionarPermiso('nunca')} 
                style={{ padding: '10px 15px', background: '#dc3545', color: 'white', border: 'none', borderRadius: '4px', cursor: 'pointer' }}
              >
                Nunca
              </button>
            </div>
          </div>
        )}

        {estadoCamara === 'activa' && (
          <div>
            <p style={{ color: 'green', fontWeight: 'bold' }}>✓ Cámara activa ({opcionPermiso})</p>
            <video ref={videoRef} autoPlay playsInline style={{ width: '100%', maxHeight: '350px', borderRadius: '6px', background: '#000' }} />
          </div>
        )}

        {estadoCamara === 'bloqueada' && (
          <div style={{ color: '#d9534f' }}>
            <p>
              {opcionPermiso === 'nunca'
                ? 'El acceso a la cámara está configurado en "Nunca".'
                : 'Acceso a la cámara denegado.'}
            </p>
          </div>
        )}
      </div>
    </div>
  );
};

// -------------------------------------------------------------
// COMPONENTE 3: Configuración de Rutas Principal (App)
// -------------------------------------------------------------
const RutaProtegida = ({ children }) => {
  const identificador = localStorage.getItem('usuario_identificador');
  return identificador ? children : <Navigate to="/login" replace />;
};

export function App() {
  return (
    <BrowserRouter>
      <Routes>
        <Route path="/login" element={<Login />} />
        
        <Route 
          path="/alumnos" 
          element={
            <RutaProtegida>
              <Alumnos />
            </RutaProtegida>
          } 
        />

        <Route path="*" element={<Navigate to="/login" replace />} />
      </Routes>
    </BrowserRouter>
  );
}

export default App;git
